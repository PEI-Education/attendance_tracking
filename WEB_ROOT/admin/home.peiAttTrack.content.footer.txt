<script>
~[if#sch.is.a.school]
  ~[if#sec.~(studentscreenaccess;attendance.html)=1]
        $j(function() {
            let requests = { ~[tlist_sql;
                select
                    count(case when u.concern_2 = 1 then 1 else null end) as letter1,
                    count(case when u.concern_3 = 1 then 1 else null end) as letter2,
                    count(case when u.concern_4 = 1 then 1 else null end) as letter3
                from u_pei_att_track u
                join students s
                    on s.dcid = u.studentsdcid
                where
                    u.termid = ~(curtermid) --Changed from yearid
                    and s.schoolid = ~(curschoolid)
                order by 1;
                ]
                    letter1: ~(1),
                    letter2: ~(2),
                    letter3: ~(3)
                [/tlist_sql]
            }
            
            let sum = Object.values(requests).reduce((a, b) => a + b, 0);
            if (sum === 1) {
                $j('h1:contains(~[text:psx.html.admin.home.start_page])').after(`<div id="attAlert" class="feedback-alert">~[text:psx.htmlc.admin_attendance.attendance_tracking.staff_have_requested_1_written] ~[text:psx.htmlc.admin_attendance.attendance_tracking.see_the_list_of_students_on_the]</div>`);
            } else if (sum > 1) {     
                $j('h1:contains(~[text:psx.html.admin.home.start_page])').after(`<div id="attAlert" class="feedback-alert">~[text:psx.htmlc.admin_attendance.attendance_tracking.staff_have_requested_multiple_written] ~[text:psx.htmlc.admin_attendance.attendance_tracking.see_the_list_of_students_on_the]</div>`);
            }
        });
  [/if#sec]
[/if#sch]
</script>
