
<script>
    $j(function() {
        let requests = { ~[tlist_sql;
                select
                    count(case when u.concern_2 = 1 then 1 else null end) as letter1,
                    count(case when u.concern_3 = 1 then 1 else null end) as letter2,
                    count(case when u.concern_4 = 1 then 1 else null end) as letter3
                from (
                    SELECT u.*,
                           ROW_NUMBER() OVER (
                               PARTITION BY u.studentsdcid, u.termid
                               ORDER BY u.whencreated ASC NULLS LAST, u.id ASC
                           ) AS checklist_rn
                    FROM u_pei_att_track u
                    WHERE u.termid = ~(curtermid)
                ) u
                join students s
                    on s.dcid = u.studentsdcid
                where
                    u.checklist_rn = 1
                    and s.schoolid = ~(curschoolid)
                    and exists (
                        select 1
                        from ps_schoolenrollment pss
                        join terms school_year
                          on school_year.schoolid = pss.schoolid
                         and school_year.yearid = ~(curyearid)
                         and school_year.id = school_year.yearid * 100
                        where pss.studentid = s.id
                          and pss.yearid = school_year.yearid
                          and pss.entrydate < school_year.lastday
                          and pss.exitdate > school_year.firstday
                          and pss.entrydate < pss.exitdate
                    );
            ]
                letter1: ~(1),
                letter2: ~(2),
                letter3: ~(3)
            [/tlist_sql]                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                               
            }
        
        let sum = Object.values(requests).reduce((a, b) => a + b, 0);
        if (sum > 0) {
           $j('#attendanceDrawerOpenButton').before(`<button id="bulkChecklists" class="btn btn-primary" onclick="window.location.assign('/admin/pei_reports/attendance_tracking.html')" style="position: relative; display: inline-block; background-color: red;"><svg class="att-icon" aria-hidden="true" focusable="false" width="1em" height="1em" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" style="vertical-align:-0.125em"><path d="M5 21V4M5 4h13l-3 4.5L18 13H5"/></svg> ~[text:psx.htmlc.admin_attendance.attendance_tracking.attendance_concerns_action_req]</button>`)
        } else {
           $j('#content-main').append(`<button id="bulkChecklists" class="btn btn-primary" onclick="window.location.assign('/admin/pei_reports/attendance_tracking.html')" style="position: absolute; right: 20px; top: 40px;"><svg class="att-icon" aria-hidden="true" focusable="false" width="1em" height="1em" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" style="vertical-align:-0.125em"><path d="M9 11a3 3 0 1 0 0-6 3 3 0 0 0 0 6zM3 20a6 6 0 0 1 12 0M16 5.2a3 3 0 0 1 0 5.6M18 14.5a6 6 0 0 1 3 5.5"/></svg> ~[text:psx.htmlc.admin_attendance.attendance_tracking.track_attendance_concerns]</button>`)
        }
    });
</script>
